title: Underhook Control
weight: 1
top_player:
  label: Top Player
  position: Top of half guard with double underhooks and chest to chest contact on your partner.
  objective: Hold bottom player down with chest to chest contact and underhooks.
  constraints: None
  win_condition: Pass the half guard by using the free leg to free the trapped leg.
bottom_player:
  label: Bottom Player
  position: Bottom of half guard with your partner having double underhooks and chest to chest contact.
  objective: Free yourself from double underhooks
  constraints: None
  win_condition: Underhooks removed.
notes: Reset if either player wins.
